# Lower-confidence patterns: a match is suspicious rather than conclusive.
#
# secrets-allow.txt suppresses these, since documentation writes
# `password: "..."` with fake values. It does not suppress secrets.txt, so an
# allowlist entry cannot wave through a real key on a line that says "example".

(?i)\b(password|passwd|secret|token|api[_-]?key|access[_-]?key|client[_-]?secret)\b\s*[:=]\s*["'][^"'\s]{12,}["']
